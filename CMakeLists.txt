cmake_minimum_required(VERSION 3.16)
project(ttl8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(ttl8-as assembler/main.cpp assembler/Assembler.cpp)
target_sources(ttl8-as PRIVATE assembler/Lexer.cpp assembler/Parser.cpp assembler/SymbolTable.cpp assembler/ObjectFile.cpp)
target_include_directories(ttl8-as PRIVATE assembler cpu)
add_library(ttl8-cpu cpu/CPU.cpp cpu/Register.cpp cpu/Bus.cpp cpu/ALU.cpp cpu/Memory.cpp cpu/InputPort.cpp cpu/OutputPort.cpp)
target_include_directories(ttl8-cpu PUBLIC cpu)
add_executable(ttl8-run cpu/main.cpp)
target_link_libraries(ttl8-run PRIVATE ttl8-cpu)
