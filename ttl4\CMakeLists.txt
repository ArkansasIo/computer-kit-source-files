cmake_minimum_required(VERSION 3.16)
project(ttl4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(ttl4-as assembler/main.cpp assembler/Assembler.cpp)
target_sources(ttl4-as PRIVATE assembler/Lexer.cpp assembler/Parser.cpp assembler/SymbolTable.cpp assembler/ObjectFile.cpp)
target_include_directories(ttl4-as PRIVATE assembler cpu)
add_library(ttl4-cpu cpu/CPU4.cpp cpu/Register4.cpp cpu/ByteRegister.cpp cpu/Bus4.cpp cpu/ALU4.cpp cpu/Memory4.cpp cpu/InputPort4.cpp cpu/OutputPort4.cpp)
target_include_directories(ttl4-cpu PUBLIC cpu)
add_executable(ttl4-run cpu/main.cpp)
target_link_libraries(ttl4-run PRIVATE ttl4-cpu)
